# LDAP-аутентификация — план реализации

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Дать питающимся вход по каталогу LDAP, автоматически заводя и обновляя их записи в Postgres, не трогая локальную аутентификацию служебных ролей.

**Architecture:** Модуль `ldap_auth` переносится внутрь приложения как `app.integrations.ldap` и остаётся чистым клиентом (bind → поиск → rebind). Логика «каталог → Postgres» живёт в сервисах: `services/ldap.py` (провайдер клиента), `services/ldap_users.py` (upsert), `services/ldap_sync.py` (ночная сверка). `services/auth.py` выбирает ветку по `User.auth_source`.

**Tech Stack:** Python 3.12, FastAPI, SQLAlchemy 2 async, Alembic, PostgreSQL 16, `ldap3`, APScheduler, React 19 + TypeScript + TanStack Query.

**Spec:** `docs/superpowers/specs/2026-10-12-ldap-auth-design.md`

## Global Constraints

- LDAP используется **только** для роли `eater`; `operator`, `admin`, `accountant` — всегда локальные.
- Пароль LDAP **не хранится и не синхронизируется** — только живой `bind` в момент входа.
- `auth_source = local` — неприкосновенен: LDAP не перехватывает вход и не меняет такие записи.
- Отключённую (`is_active = false`) LDAP-запись автоматически обратно **не включаем**.
- При создании питающегося назначается **первый активный зал по алфавиту**; активных залов нет → `503 no_hall_available`.
- Каталог недоступен → `503 ldap_unavailable` на входе; при синке — предупреждение в лог, изменений нет.
- `LDAP_ENABLED=false` (дефолт) → поведение полностью прежнее.
- Синк: `LDAP_SYNC_HOUR:59`, дефолт `LDAP_SYNC_HOUR=23`. Пустой фильтр импорта → прогон не выполняется.
- Секреты только в `.env`; `ldap_module/.env` в git не попадает и **не удаляется**.
- UI — по-русски, без эмодзи; только дизайн-токены (`bg-surface`, `border-border`, `text-muted`, `min-h-11`, `text-on-accent`).

```python
# server/app/services/ldap_users.py
from __future__ import annotations

import logging
from dataclasses import dataclass

from sqlalchemy.ext.asyncio import AsyncSession

from app.integrations.ldap import LdapIdentity
from app.models import AuthSource, User, UserHall, UserRole
from app.repositories import halls as halls_repo
from app.repositories import users as users_repo

logger = logging.getLogger(__name__)


class NoHallAvailable(RuntimeError):
    """Активных залов нет — нового питающегося создать нельзя."""


@dataclass
class UpsertResult:
    user: User | None
    created: bool = False
    skipped: str | None = None  # "inactive" | "login_taken"


async def upsert_ldap_user(
    db: AsyncSession, identity: LdapIdentity, *, source: str = "login"
) -> UpsertResult:
    """Создаёт или обновляет локальную запись по данным каталога.

    ``skipped == "inactive"`` — запись отключена оператором, обратно не включаем.
    ``skipped == "login_taken"`` — логин занят локальной учёткой.
    """
    login = identity.login.strip().lower()
    uid = identity.uid[:128]
    dn = identity.dn[:512]
    full_name = (identity.full_name or login).strip()[:255] or login
    if not login:
        return UpsertResult(user=None, skipped="login_taken")

    user = await users_repo.get_by_ldap_uid(db, uid)
    if user is not None:
        if not user.is_active:
            logger.info("LDAP(%s): %s отключён локально — пропуск", source, login)
            return UpsertResult(user=None, skipped="inactive")
        user.full_name = full_name
        user.ldap_dn = dn
        if user.login != login:
            holder = await users_repo.get_by_login(db, login)
            if holder is None or holder.id == user.id:
                user.login = login
            else:
                logger.warning(
                    "LDAP(%s): новый логин %s занят — у %s оставлен %s",
                    source, login, uid, user.login,
                )
        await db.flush()
        return UpsertResult(user=user)

    existing = await users_repo.get_by_login(db, login)
    if existing is not None:
        if existing.auth_source == AuthSource.local:
            logger.warning(
                "LDAP(%s): логин %s занят локальной учёткой — пропуск", source, login
            )
            return UpsertResult(user=None, skipped="login_taken")
        if not existing.is_active:
            logger.info("LDAP(%s): %s отключён локально — пропуск", source, login)
            return UpsertResult(user=None, skipped="inactive")
        existing.ldap_uid = uid
        existing.ldap_dn = dn
        existing.full_name = full_name
        await db.flush()
        return UpsertResult(user=existing)

    hall = await halls_repo.first_active(db)
    if hall is None:
        raise NoHallAvailable(login)

    user = User(
        login=login,
        password_hash=None,
        full_name=full_name,
        role=UserRole.eater,
        auth_source=AuthSource.ldap,
        ldap_uid=uid,
        ldap_dn=dn,
        is_active=True,
    )
    db.add(user)
    await db.flush()
    db.add(UserHall(user_id=user.id, hall_id=hall.id))
    await db.flush()
    logger.info("LDAP(%s): создан %s (%s)", source, login, hall.name)
    return UpsertResult(user=user, created=True)
```

```python
    def identity_from_entry(self, entry: DirectoryEntry) -> LdapIdentity | None:
        """Собирает учётную запись из записи каталога (для синка)."""
        login = _as_text(entry.attributes.get(self._settings.login_attr)).strip().lower()
        if not login:
            logger.warning(
                "LDAP: у записи %s нет атрибута %s — пропуск",
                entry.dn,
                self._settings.login_attr,
            )
            return None
        uid = _as_text(entry.attributes.get(self._settings.id_attr)) or entry.dn
        full_name = _as_text(entry.attributes.get(self._settings.name_attr)) or login
        return LdapIdentity(uid=uid, dn=entry.dn, login=login, full_name=full_name)

    def list_entries_sync(self, search_filter: str) -> list[DirectoryEntry]:
        """Ищет все записи под ``LDAP_BASE_DN`` сервисной учёткой."""
        base_dn = self._settings.base_dn
        bind_dn = self._settings.bind_dn
        bind_password = self._settings.bind_password
        if not base_dn or not bind_dn or bind_password is None:
            raise LdapUnavailable("LDAP настроен не полностью")

        connection = self._connector.connect()
        try:
            if not connection.bind(bind_dn, bind_password):
                raise LdapUnavailable("сервисная учётка LDAP не прошла bind")
            return connection.search(
                base_dn,
                search_filter,
                attributes=[
                    self._settings.login_attr,
                    self._settings.name_attr,
                    self._settings.id_attr,
                ],
                size_limit=0,
            )
        finally:
            connection.unbind()

    async def list_entries(self, search_filter: str) -> list[DirectoryEntry]:
        return await asyncio.to_thread(self.list_entries_sync, search_filter)
```

```python
    def summary(self) -> str:
        return (
            f"создано {self.created}, обновлено {self.updated}, "
            f"отключено {self.deactivated}, пропущено {self.skipped}"
        )


async def sync_directory(db: AsyncSession) -> SyncResult:
    """Полная сверка каталога и локальных LDAP-записей.

    :raises SyncFilterMissing: пустой `LDAP_SYNC_FILTER`/`LDAP_USER_FILTER`.
    :raises LdapUnavailable: каталог недоступен; изменений не применяем.
    """
    search_filter = ldap_service.get_ldap_settings().import_filter()
    if not search_filter:
        raise SyncFilterMissing(
            "LDAP_SYNC_FILTER и LDAP_USER_FILTER пусты — импорт всего каталога запрещён"
        )

    identities = await ldap_service.list_identities(search_filter)
    seen: set[str] = set()
    stats = SyncResult()

    for identity in identities:
        uid = identity.uid[:128]
        result = await ldap_users.upsert_ldap_user(db, identity, source="sync")
        seen.add(uid)
        if result.user is None:
            if result.skipped == "login_taken":
                stats.skipped += 1
            continue
        if result.created:
            stats.created += 1
        else:
            stats.updated += 1

    for user in await users_repo.list_ldap(db, only_active=True):
        if user.ldap_uid and user.ldap_uid not in seen:
            user.is_active = False
            stats.deactivated += 1

    await db.flush()
    logger.info("LDAP-синк: %s", stats.summary())
    return stats
```
